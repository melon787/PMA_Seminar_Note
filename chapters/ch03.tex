\section{Numerical Sequences and Series}
\subsection{Convergent Sequences}

\begin{definition}
A sequence $\{p_n\}$ in a metric space $X$ is said to \emph{converge} if there is a point $p \in X$ with the following property: 
For every $\varepsilon>0$ there is an integer $N$ such that $n \geq N$ implies that $d(p_n,p)<\varepsilon$. 
(Here $d$ denotes the distance in $X$.)

In this case we also say that $\{p_n\}$ converges to $p$, or that $p$ is the limit of $\{p_n\}$ [see Theorem \ref{thm:limit_property}(b)], and we write $\{p_n\} \to p$, or
\[
\lim_{n \to \infty} p_n = p.
\]

If $\{p_n\}$ does not converge, it is said to \emph{diverge}.

We recall that the set of all points $p_n$ ($n=1, 2, 3, \ldots$) is  the \emph{range} of $\{p_n\}$.
The range of a sequence may be a finite set, or it may be infinite.
The sequence $\{p_n\}$ is said to be \emph{bounded} if its range is bounded.
\end{definition}

\begin{theorem} \label{thm:limit_property}
Let $\{p_n\}$ be a sequence in a metric space $X$.

\begin{enumerate}
    \item[(a)] $\{p_n\}$ converges to $p \in X$ if and only if every neighborhood of $p$ contains $p_n$ for all but finitely many $n$.
    \item[(b)] If $p \in X$, $p' \in X$, and if $\{p_n\}$ converges to $p$ and to $p'$, then $p'=p$.
    \item[(c)] If $\{p_n\}$ converges, then $\{p_n\}$ is bounded.
    \item[(d)] If $E \subset X$ and if $p$ is a limit point of $E$, then there is a sequence $\{p_n\}$ in $E$ such that $p=\lim\limits_{n \to \infty} p_n$.
\end{enumerate}
\end{theorem}

\begin{proof}
Since Rudin's proofs of parts (a), (b), and (c) are already explicit, we omit them here.

We prove part (d). 
Since $p$ is a limit point of $E$, for $n=1, 2, 3, \ldots$, choose a point $p_n \in E$ such that $d(p_n,p)<1/n$.
Given $\varepsilon>0$, there exists a positive integer $N$ such that $N\varepsilon>1$.
Then $n \geq N$ implies 
\[
d(p_n,p)<\frac{1}{n}\leq\frac{1}{N}<\varepsilon.
\]
Therefore, $\{p_n\}$ converges to $p$.
\end{proof}

\begin{theorem}
Suppose $\{s_n\}$, $\{t_n\}$ are complex sequences, and $\lim_{n \to \infty} s_n =s$, $\lim_{n \to \infty} t_n =t$.
Then
\begin{enumerate}
    \item[(a)] $\lim\limits_{n \to \infty} (s_n + t_n) = s+t$;
    \item[(b)] $\lim\limits_{n \to \infty} cs_n=cs$, $\lim\limits_{n \to \infty}(c+s_n)=c+s$, for any number $c$;
    \item[(c)] $\lim\limits_{n \to \infty} s_nt_n=st$;
    \item[(d)] $\lim\limits_{n \to \infty} \displaystyle\frac{1}{s_n}=\frac{1}{s}$, provided $s_n \neq 0$ ($n=1, 2, 3, \ldots$), and $s \neq 0$.
\end{enumerate} 
\end{theorem}

\begin{theorem} \,
\begin{enumerate}
    \item[(a)] Suppose $\mathbf{x}_n \in \R^k$ ($n=1, 2, 3, \ldots$) and \[\mathbf{x}_n=(\alpha_{1,n}, \ldots, \alpha_{k,n}).\] 
    Then $\{\mathbf{x}_n\}$ converges to $\mathbf{x}=(\alpha_1, \ldots, \alpha_k)$ if and only if
    \begin{equation}
        \lim_{n \to \infty} \alpha_{j,n} = \alpha_j \qquad (1 \leq j \leq k).
    \end{equation}
    \item[(b)] Suppose $\{\mathbf{x}_n\}$, $\{\mathbf{y}_n\}$ are sequences in $\R^k$, $\{\beta_n\}$ is a sequence of real numbers, and $\mathbf{x}_n \to \mathbf{x}$, $\mathbf{y}_n \to \mathbf{y}$, $\beta_n \to \beta$.
    Then
    \[
    \lim_{n \to \infty} (\mathbf{x}_n + \mathbf{y}_n) = \mathbf{x} + \mathbf{y}, \qquad \lim_{n \to \infty} \mathbf{x}_n \cdot \mathbf{y}_n = \mathbf{x} \cdot \mathbf{y}, \qquad \lim_{n \to \infty} \beta_n \mathbf{x}_n = \beta \mathbf{x}.
    \]
\end{enumerate}
\end{theorem}

\subsection{Subsequences}

\begin{definition} \label{def:subsequence}
Given a sequence $\{p_n\}$, consider a sequence $\{n_k\}$ of positive integers, such that $n_1 < n_2 < n_3 < \cdots$.
Then the sequence $\{p_{n_i}\}$ is called a \emph{subsequence} of $\{p_n\}$.
If $\{p_{n_i}\}$ converges, its limit is called a \emph{subsequential limit} of $\{p_n\}$.
\end{definition}


A sequence $\{p_n\}$ converges to $p$ if and only if every subsequence of $\{p_n\}$ converges to $p$.

To prove that, suppose $\{p_n\}$ converges to $p$ and let $\{p_{n_i}\}$ be a subsequence of $\{p_n\}$. 
Given $\varepsilon>0$, there exists a positive integer $N$ such that $d(p_n,p)<\varepsilon$ for $n \geq N$.
Since $n_i \geq i$ for $i=1, 2, 3, \ldots$, $i \geq N$ implies $n_i \geq N$ and therefore $d(p_{n_i},p)<\varepsilon$.
Hence $\{p_{n_i}\}$ converges to $p$.

The converse is trivial.

\begin{theorem} \,
    \begin{enumerate}
        \item[(a)] If $\{p_n\}$ is a sequence in a compact metric space $X$, then some subsequence of $\{p_n\}$ converges to a point of $X$.
        \item[(b)] Every bounded sequence in $\R^k$ contains a convergent subsequence.
    \end{enumerate}
\end{theorem}

\begin{proof}
\,
\begin{enumerate}
\item[(a)] Let $E$ be a range of $\{p_n\}$.
If $E$ is finite then there are infinitely many terms such that
    \[
    p_{n_1}=p_{n_2}=p_{n_3}=\cdots=p.
    \]
Without loss of generality, we may assume that $n_1<n_2<n_3< \cdots$.
It is clear that $\{p_{n_i}\}$ converges to $p$.

If $E$ is infinite, Theorem \ref{thm:compact_limit} shows that $E$ has a limit point $p \in X$.
Choose $n_1$ so that $d(p,p_{n_1})<1$.
Having chosen $n_1, \ldots, n_{i-1}$, we see from Theorem \ref{thm:limit_cluster} that there is an integer $n_i>n_{i-1}$ such that $d(p,p_{n_i})<1/i$.
Then $\{p_{n_i}\}$ converges to $p$.

\item[(b)] This follows from (a) and the Heine--Borel Theorem.
\end{enumerate}
\end{proof}

\begin{theorem} \label{thm:subseq_closed}
The subsequential limits of a sequence $\{p_n\}$ in a metric space $X$ form a closed subset of $X$.
\end{theorem}

\subsection{Cauchy Sequences}

\begin{definition}
A sequence $\{p_n\}$ in a metric space $X$ is said to be a \emph{Cauchy sequence} if for every $\varepsilon>0$ there is an integer $N$ such that $d(p_n,p_m)<\varepsilon$ if $n \geq N$ and $m \geq N$.
\end{definition}

\begin{definition}
Let $E$ be a nonempty subset of a metric space $X$, and let $S$ be the set of all real numbers of the form $d(p,q)$, with $p \in E$ and $q \in E$.
The sup of $S$ is called the \emph{diameter} of $E$.
\end{definition}

Let $\{p_n\}$ be a sequence in $X$ and let $E_N$ consist of the points $p_N, p_{N+1}, p_{N+2}, \ldots$. 
We prove that \emph{$\{p_n\}$ is a Cauchy sequence if and only if}
\[
\lim_{N \to \infty} \diam E_N = 0.
\]

Suppose $\{p_n\}$ is Cauchy.
Given $\varepsilon>0$, choose an integer $N_0$ such that $n,m \geq N_0$ imply $d(p_n,p_m)<\varepsilon$.
It follows that $\diam E_{N_0} \leq \varepsilon$.
Since $E_{N_0} \supset E_{N_0+1} \supset E_{N_0+2} \supset \cdots$, we have $\diam E_{N_0} \geq \diam E_{N_0+1} \geq \diam E_{N_0+2} \geq \cdots$.
Therefore, $N \geq N_0$ implies $\diam E_N \leq \varepsilon$. 
This proves that $\lim_{N \to \infty} \diam E_N = 0$.

Conversely, suppose $\lim_{N \to \infty} \diam E_N = 0$.
For every $\varepsilon>0$, there exists an integer $N_1$ such that $\diam E_{N_1} <\varepsilon$.
Thus, if $n \geq N_1$ and $m \geq N_1$ then $d(p_n,p_m) \leq \diam E_{N_1} < \varepsilon$. 
Hence $\{p_n\}$ is a Cauchy sequence.
\begin{theorem}\,
\begin{enumerate}
\item [(a)] If $\overline{E}$ is the closure of a set $E$ in a metric space $X$, then
\[
\diam \overline{E} = \diam E.
\]
\item[(b)] If $K_n$ is a sequence of compact sets in $X$ such that $K_n \supset K_{n+1}$ ($n=1,2,3,\ldots$) and if
\[
\lim_{n \to \infty} \diam K_n = 0,
\]
then $\cap_{1}^{\infty}K_n$ consists of exactly one point.
\end{enumerate}
\end{theorem}

\begin{theorem}\label{thm:compact_completeness} \,
\begin{enumerate}
\item[(a)] In any metric space $X$, every convergent sequence is a Cauchy sequence.
\item[(b)] If $X$ is a compact metric space and if $\{p_n\}$ is a Cauchy sequence in $X$, then $\{p_n\}$ converges to some point of $X$.
\item[(c)] In $\R^k$, every Cauchy sequence converges.
\end{enumerate}
\end{theorem}

\begin{definition}
A metric space in which every Cauchy sequence converges is said to be \emph{complete}.
\end{definition}

Thus Theorem \ref{thm:compact_completeness} says that \emph{all compact metric spaces and all Euclidean spaces are complete}.
Theorem \ref{thm:compact_completeness} implies also that \emph{every closed subset $E$ of a complete metric space $X$ is complete}. \footnote{This result is also immediate from definition.}

Theorem \ref{thm:limit_property}(c) shows that convergent sequences are bounded.
But bounded sequences need not converge.
However, there is one important case in which covnergence \emph{is} equivalent to boundness.

\begin{definition}
A sequence $\{s_n\}$ of real numbers is said to be

\begin{enumerate}
    \item[(a)] \emph{monotonically increasing} if $s_n \leq s_{n+1}$ ($n=1, 2, 3, \ldots$);
    \item[(b)] \emph{monotonically decreasing} if $s_n \geq s_{n+1}$ ($n=1, 2, 3, \ldots$). 
\end{enumerate}
\end{definition}

\begin{theorem}
Suppose $\{s_n\}$ is monotonic.
Then $\{s_n\}$ converges if and only if it is bounded.
\end{theorem}

\subsection{Upper and Lower Limits}
\begin{definition} 
Let $\{s_n\}$ be a sequence of real numbers with the following property:
For every real $M$ there is an integer $N$ such that $n \geq N$ implies $s_n \geq M$.
We then write
\[
s_n \to +\infty.
\]
Similarly, if for every real $M$ there is an integer $N$ such that $n \geq N$ implies $s_n \leq M$, we write
\[
s_n \to -\infty.
\]
\end{definition}

\begin{definition}\label{def:upper_lower_limits}
Let $\{s_n\}$ be a sequence of real numbers.
Let $E$ be the set of numbers $x$ (in the extended real number system) such that $s_{n_k} \to x$ for some subsequence $\{s_{n_k}\}$.
This set $E$ contains all subsequential limits as defined in Definition \ref{def:subsequence}, plus possibly the numbers $+ \infty$, $- \infty$.

We now recall Definitions \ref{def:sup_inf} and \ref{def:infinity} and put
\begin{align*}
    s^{*} = \sup E,\\
    s_{*} = \inf E.
\end{align*}
The numbers $s^*$, $s_*$ are called the \emph{upper} and \emph{lower limits} of $\{s_n\}$;
we use the notation
\[
\limsup_{n \to \infty}s_n = s^*, \qquad \liminf_{n \to \infty}s_n=s_*.
\]
\end{definition}

\begin{theorem}
Let $\{s_n\}$ be a sequence of real numbers.
Let $E$ and $s^*$ have the same meaning as in Definition \ref{def:upper_lower_limits}.
Then $s^*$ has the following two properties:
\begin{enumerate}
    \item[(a)] $s^* \in E$.
    \item[(b)] If $x>s^*$, there is an integer $N$ such that $n \geq N$ implies $s_n<x$.
\end{enumerate}

Moreover, $s^*$ is the only number with the properties (a) and (b).
\end{theorem}

Of course, an analogous result is true for $s_*$.

\begin{proof}
Since Rudin's proofs of part (b) and the uniqueness of $s^*$ are already explicit, we omit them here.
We prove part (a).

If $s^*=+\infty$ then by the definition of supremum, no real number is an upper bound of $E$.
Hence $E$ is not bounded above in $\R$.
It follows that $\{s_n\}$ is not bounded above, for otherwise $\sup E$ would exist in $\R \cup \{-\infty\}$.
Pick an integer $n_1$ such that $s_{n_1}>1$.
Having chosen $n_1, \ldots, n_{k-1}$, there exists an $n_k>n_{k-1}$ such that $s_{n_k}>k$.
Thus, the last inequality holds for $k=1,2,3, \ldots$ and this implies that $s_{n_k} \to +\infty$.

If $s^*$ is real, this means that $\sup E$ exists in $\R$.
Removing $-\infty$ from $E$ if necessary\footnote{This does not change the supremum of $E$. See Remark \ref{removing_minus_infinity}.}, we may assume that $E$ is a subset of $\R$.
Then $s^*= \sup E \in \overline{E}$.
Hence (a) follows from Theorem \ref{thm:subseq_closed}.

If $s^*=-\infty$, then $s^*=\sup E<x$ for every real number $x$. 
Hence there is no subsequential limit of $\{s_n\}$ (in $\R$) and $E$ contains at most one element, namely $- \infty$.
If $\{s_n\}$ were bounded, then there would be a convergent subsequence $\{s_{n_k}\}$.
This says that $\{s_n\}$ is not bounded.
Since there is no subsequence of $\{s_n\}$ which diverges to $+\infty$, $\{s_n\}$ is bounded above.
Hence $\{s_n\}$ is not bounded below and it follows that $s^*=-\infty \in E$.

This establishes (a) in all cases.
\end{proof}

\begin{remark} \label{removing_minus_infinity}
Let $S$ be an ordered set and $E$ be a subset of $S$. 
Suppose
\[
E=\{a\} \cup \widetilde{E}
\]
where $a$ is not an upper bound of $\widetilde{E}$. 
If the suprema of $E$ and $\widetilde{E}$ exist in $S$, then $\sup \widetilde{E} = \sup E$.
\end{remark}

\begin{proof}
Since $\widetilde{E} \subset E$, we have $\sup \widetilde{E} \leq \sup E$.
Suppose $ \sup \widetilde{E} < \sup E$.
Then there exists an $x \in E$ such that $\sup \widetilde{E} < x$.
Hence, $x \notin \widetilde{E}$, that is, $x=a$.
But $a$ is not an upper bound of $\widetilde{E}$ by hypothesis, a contradiction.
\end{proof}

\begin{theorem}
If $s_n \leq t_n$ for $n \geq N$, where $N$ is fixed, then
\begin{align*}
    \liminf_{n \to \infty}s_n \leq \liminf_{n \to \infty}t_n, \\
    \limsup_{n \to \infty}s_n \leq \limsup_{n \to \infty} t_n.
\end{align*}
\end{theorem}

\subsection*{Exercises}
\addcontentsline{toc}{subsection}{Exercises}

\begin{exercise}
Suppose $a_n>0$, $s_n=a_1+\cdots+a_n$, and $\sum a_n$ diverges.
\begin{enumerate}
    \item[(a)] Prove that $\sum\frac{a_n}{1+a_n}$ diverges.
    \item[(b)] Prove that 
\[
\frac{a_{N+1}}{s_{N+1}} + \cdots + \frac{a_{N+k}}{s_{N+k}} \geq 1- \frac{s_{N}}{s_{N+k}}
\]
and deduce that $\sum \frac{a_n}{s_n}$ diverges.
    \item[(c)] Prove that
\[
\frac{a_n}{s_n^2} \leq \frac{1}{s_{n-1}}-\frac{1}{s_n}
\]
and deduce that $\sum \frac{a_n}{s_n^2}$ converges.
    \item[(d)] What can be said about
\[
\sum \frac{a_n}{1+na_n} \text{ and } \sum \frac{a_n}{1+n^2a_n} ?
\]
\end{enumerate}
\end{exercise}
\begin{proof}
\begin{enumerate}
    \item[(a)] Assume $a_n$ diverges. 
If $\{n_i\}$ is a sequence of positive integers with $n_1<n_2<n_3<\cdots$, then we have
\[
\sum_{i=1}^{n_k} \frac{a_i}{1+a_i} \geq \sum_{i=1}^k \frac{a_{n_i}}{1+a_{n_i}}.
\]

Since $a_n$ diverges, we have $0<s=\limsup_{n \to \infty} a_n \leq \infty$. 
Pick a subsequence $\{a_{n_i}\}_{i \in \N}$ such that $a_{n_i} \to s$ and let $k \to \infty$.
Then right side of the last inequality diverges, since $\frac{a_{n_i}}{1+a_{n_i}}$ does not converge to $0$.
Hence left hand side diverges.

Next, assume $a_n$ converges and let $a_n \to 0$. (Otherwise, $\sum \frac{a_n}{1+a_n}$ diverges.)
Then there exists an integer $N_1$ such that $|a_n|<1/2$ for $n \geq N_1$.

Suppose, for a contradiction, that $\sum \frac{a_n}{1+a_n}$ converges.
Let $\varepsilon>0$ be given. 
Choose an integer $N_2$ such that 
\[
m \geq n \geq N_2 \text{ implies } \sum_{k=n}^m\frac{a_k}{1+a_k}  \leq \varepsilon/2.
\]

Now, let $N= \max \{N_1,N_2\}$ and let $\alpha= \sup_{k \geq N} a_k$.
If $m \geq n \geq N$ then
\begin{align*}
    \frac{a_n}{1+a_n}+ \cdots +\frac{a_m}{1+a_m}  &\geq \frac{1}{1+\alpha}(a_n+ \cdots +a_m) \\
    &=\frac{1}{1+\alpha} \sum_{k=n}^m a_k,
\end{align*}
hence 
\[
\sum_{k=n}^m a_k \leq \frac{1+\alpha}{2} \varepsilon \leq \varepsilon.
\]
This contradicts our hypothesis that $\sum a_n$ diverges.

\item[(b)] Since $a_n>0$, we have
\begin{align*}
\frac{a_{N+1}}{s_{N+1}}+ \cdots +\frac{a_{N+k}}{s_{N+k}} &\geq \frac{a_{N+1}+ \cdots a_{N+k}}{s_{N+k}} \\
&=\frac{s_{N+k}-s_N}{s_{N+k}} \\
&= 1-\frac{s_N}{s_{N+k}}.
\end{align*}
for every positive integer $N$.
This yields the desired inequality.

Suppose $\sum \frac{a_n}{s_n}$ converges.
Choose an integer $N$ such that $m \geq n \geq N$ implies 
\[
\sum_{k=n}^m \frac{a_k}{s_k} \leq 1/2.
\]
Hence we have
\[
1-\frac{s_N}{s_{N+k}} \leq \frac{a_{N+1}}{s_{N+1}} + \cdots + \frac{a_{N+k}}{s_{N+k}}
\]
for fixed $N$ and for all $k \in \N$.
Letting $k \to \infty$, we obtain $1 \leq \frac{1}{2}$, a contradiction.

\item[(c)] Observe that
\begin{align*}
    \frac{a_n}{s_n^2} &\leq \frac{a_n}{s_{n-1}s_n} \\
    &= \frac{a_n}{s_n-s_{n-1}}\left(\frac{1}{s_{n-1}}-\frac{1}{s_n}\right) \\
    &=\frac{1}{s_{n-1}}-\frac{1}{s_n}.
\end{align*}
Taking summation from $n=2$ of both sides and letting $n \to \infty$, by telescoping, we conclude that $\sum \frac{a_n}{s_n^2}$ converges.

\item[(d)] We first prove that the latter one converges.
Observe that
\[
\frac{a_n}{1+n^2a_n} \leq \frac{a_n}{n^2a_n}=\frac{1}{n^2},
\]
which implies that $\sum \frac{a_n}{1+n^2a_n}$ converges.

However, there is nothing that we can say about the former one.
For instance, if we put $a_n=\frac{1}{n}$ then $\sum \frac{a_n}{1+na_n}=\sum \frac{1}{2n}$ diverges. 
But if we take $a_{m^2}=1/m$ for $m=1,2,3, \ldots$, while $a_n=2^{-n}$ if $n$ is not a square, then the sum converges.\footnote{The last example is from this: https://math.stackexchange.com/questions/237691/baby-rudin-chapter-3-exercise-11d}
\end{enumerate}
\end{proof}

\begin{exercise}
suppose $\{p_n\}$ is a Cauchy sequence in a metric space $X$, and some subsequence $\{p_{n_i}\}$ converges to a point $p \in X$.
Prove that the full sequence $\{p_n\}$ converges to $p$.
\end{exercise}
\begin{proof}
Fix $\varepsilon>0$.
Since $\{p_n\}$ is Cauchy, there exists an integer $N$ such that $d(p_n,p_m)<\varepsilon/2$ if $n \geq N$ and $m \geq N$.
Since a subsequence $\{p_{n_i}\}$ converges to $p$, choose an integer $k \geq N$ such that $d(p_{n_k},p)<\varepsilon/2$.
Hence, $n \geq N$ implies
\begin{align*}
d(p_n,p) &\leq d(p_n, p_{n_k})+d(p_{n_k}, p) \\
&<\varepsilon/2+\varepsilon/2=\varepsilon.
\end{align*}
This is the desired result.
\end{proof}
