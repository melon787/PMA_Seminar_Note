\section{Numerical Sequences and Series}
\subsection{Convergent Sequences}

\begin{definition}
A sequence $\{p_n\}$ in a metric space $X$ is said to \emph{converge} if there is a point $p \in X$ with the following property: 
For every $\varepsilon>0$ there is an integer $N$ such that $n \geq N$ implies that $d(p_n,p)<\varepsilon$. 
(Here $d$ denotes the distance in $X$.)

In this case we also say that $\{p_n\}$ converges to $p$, or that $p$ is the limit of $\{p_n\}$ [see Theorem \ref{thm:limit_property}(b)], and we write $\{p_n\} \to p$, or
\[
\lim_{n \to \infty} p_n = p.
\]

If $\{p_n\}$ does not converge, it is said to \emph{diverge}.

We recall that the set of all points $p_n$ ($n=1, 2, 3, \ldots$) is  the \emph{range} of $\{p_n\}$.
The range of a sequence may be a finite set, or it may be infinite.
The sequence $\{p_n\}$ is said to be \emph{bounded} if its range is bounded.
\end{definition}

\begin{theorem} \label{thm:limit_property}
Let $\{p_n\}$ be a sequence in a metric space $X$.

\begin{enumerate}
    \item[(a)] $\{p_n\}$ converges to $p \in X$ if and only if every neighborhood of $p$ contains $p_n$ for all but finitely many $n$.
    \item[(b)] If $p \in X$, $p' \in X$, and if $\{p_n\}$ converges to $p$ and to $p'$, then $p'=p$.
    \item[(c)] If $\{p_n\}$ converges, then $\{p_n\}$ is bounded.
    \item[(d)] If $E \subset X$ and if $p$ is a limit point of $E$, then there is a sequence $\{p_n\}$ in $E$ such that $p=\lim\limits_{n \to \infty} p_n$.
\end{enumerate}
\end{theorem}

\begin{proof}
Since Rudin's proofs of parts (a), (b), and (c) are already explicit, we omit them here.

We prove part (d). 
Since $p$ is a limit point of $E$, for $n=1, 2, 3, \ldots$, choose a point $p_n \in E$ such that $d(p_n,p)<1/n$.
Given $\varepsilon>0$, there exists a positive integer $N$ such that $N\varepsilon>1$.
Then $n \geq N$ implies 
\[
d(p_n,p)<\frac{1}{n}\leq\frac{1}{N}<\varepsilon.
\]
Therefore, $\{p_n\}$ converges to $p$.
\end{proof}

\begin{theorem}
Suppose $\{s_n\}$, $\{t_n\}$ are complex sequences, and $\lim_{n \to \infty} s_n =s$, $\lim_{n \to \infty} t_n =t$.
Then
\begin{enumerate}
    \item[(a)] $\lim\limits_{n \to \infty} (s_n + t_n) = s+t$;
    \item[(b)] $\lim\limits_{n \to \infty} cs_n=cs$, $\lim\limits_{n \to \infty}(c+s_n)=c+s$, for any number $c$;
    \item[(c)] $\lim\limits_{n \to \infty} s_nt_n=st$;
    \item[(d)] $\lim\limits_{n \to \infty} \displaystyle\frac{1}{s_n}=\frac{1}{s}$, provided $s_n \neq 0$ ($n=1, 2, 3, \ldots$), and $s \neq 0$.
\end{enumerate} 
\end{theorem}

\begin{theorem} \,
\begin{enumerate}
    \item[(a)] Suppose $\mathbf{x}_n \in \R^k$ ($n=1, 2, 3, \ldots$) and \[\mathbf{x}_n=(\alpha_{1,n}, \ldots, \alpha_{k,n}).\] 
    Then $\{\mathbf{x}_n\}$ converges to $\mathbf{x}=(\alpha_1, \ldots, \alpha_k)$ if and only if
    \begin{equation}
        \lim_{n \to \infty} \alpha_{j,n} = \alpha_j \qquad (1 \leq j \leq k).
    \end{equation}
    \item[(b)] Suppose $\{\mathbf{x}_n\}$, $\{\mathbf{y}_n\}$ are sequences in $\R^k$, $\{\beta_n\}$ is a sequence of real numbers, and $\mathbf{x}_n \to \mathbf{x}$, $\mathbf{y}_n \to \mathbf{y}$, $\beta_n \to \beta$.
    Then
    \[
    \lim_{n \to \infty} (\mathbf{x}_n + \mathbf{y}_n) = \mathbf{x} + \mathbf{y}, \qquad \lim_{n \to \infty} \mathbf{x}_n \cdot \mathbf{y}_n = \mathbf{x} \cdot \mathbf{y}, \qquad \lim_{n \to \infty} \beta_n \mathbf{x}_n = \beta \mathbf{x}.
    \]
\end{enumerate}
\end{theorem}

\subsection{Subsequences}

\begin{definition}
Given a sequence $\{p_n\}$, consider a sequence $\{n_k\}$ of positive integers, such that $n_1 < n_2 < n_3 < \cdots$.
Then the sequence $\{p_{n_i}\}$ is called a \emph{subsequence} of $\{p_n\}$.
If $\{p_{n_i}\}$ converges, its limit is called a \emph{subsequential limit} of $\{p_n\}$.
\end{definition}


A sequence $\{p_n\}$ converges to $p$ if and only if every subsequence of $\{p_n\}$ converges to $p$.

To prove that, suppose $\{p_n\}$ converges to $p$ and let $\{p_{n_i}\}$ be a subsequence of $\{p_n\}$. 
Given $\varepsilon>0$, there exists a positive integer $N$ such that $d(p_n,p)<\varepsilon$ for $n \geq N$.
Since $n_i \geq i$ for $i=1, 2, 3, \ldots$, $i \geq N$ implies $n_i \geq N$ and therefore $d(p_{n_i},p)<\varepsilon$.
Hence $\{p_{n_i}\}$ converges to $p$.

The converse is trivial.

\begin{theorem} \,
    \begin{enumerate}
        \item[(a)] If $\{p_n\}$ is a sequence in a compact metric space $X$, then some subsequence of $\{p_n\}$ converges to a point of $X$.
        \item[(b)] Every bounded sequence in $\R^k$ contains a convergent subsequence.
    \end{enumerate}
\end{theorem}

\begin{proof}
\,
\begin{enumerate}
\item[(a)] Let $E$ be a range of $\{p_n\}$.
If $E$ is finite then there are infinitely many terms such that
    \[
    p_{n_1}=p_{n_2}=p_{n_3}=\cdots=p.
    \]
Without loss of generality, we may assume that $n_1<n_2<n_3< \cdots$.
It is clear that $\{p_{n_i}\}$ converges to $p$.

If $E$ is infinite, Theorem \ref{thm:compact_limit} shows that $E$ has a limit point $p \in X$.
Choose $n_1$ so that $d(p,p_{n_1})<1$.
Having chosen $n_1, \ldots, n_{i-1}$, we see from Theorem \ref{thm:limit_cluster} that there is an integer $n_i>n_{i-1}$ such that $d(p,p_{n_i})<1/i$.
Then $\{p_{n_i}\}$ converges to $p$.

\item[(b)] This follows from (a) and the Heine--Borel Theorem.
\end{enumerate}
\end{proof}

\begin{theorem}
The subsequential limits of a sequence $\{p_n\}$ in a metric space $X$ form a closed subset of $X$.
\end{theorem}

\subsection{Cauchy Sequences}

\begin{definition}
A sequence $\{p_n\}$ in a metric space $X$ is said to be a \emph{Cauchy sequence} if for every $\varepsilon>0$ there is an integer $N$ such that $d(p_n,p_m)<\varepsilon$ if $n \geq N$ and $m \geq N$.
\end{definition}

\begin{definition}
Let $E$ be a nonempty subset of a metric space $X$, and let $S$ be the set of all real numbers of the form $d(p,q)$, with $p \in E$ and $q \in E$.
The sup of $S$ is called the \emph{diameter} of $E$.
\end{definition}

Let $\{p_n\}$ be a sequence in $X$ and let $E_N$ consist of the points $p_N, p_{N+1}, p_{N+2}, \ldots$. 
We prove that \emph{$\{p_n\}$ is a Cauchy sequence if and only if}
\[
\lim_{N \to \infty} \diam E_N = 0.
\]

Suppose $\{p_n\}$ is Cauchy.
Given $\varepsilon>0$, choose an integer $N_0$ such that $n,m \geq N_0$ imply $d(p_n,p_m)<\varepsilon$.
It follows that $\diam E_{N_0} \leq \varepsilon$.
Since $E_{N_0} \supset E_{N_0+1} \supset E_{N_0+2} \supset \cdots$, we have $\diam E_{N_0} \geq \diam E_{N_0+1} \geq \diam E_{N_0+2} \geq \cdots$.
Therefore, $N \geq N_0$ implies $\diam E_N \leq \varepsilon$. 
This proves that $\lim_{N \to \infty} \diam E_N = 0$.

Conversely, suppose $\lim_{N \to \infty} \diam E_N = 0$.
For every $\varepsilon>0$, there exists an integer $N_1$ such that $\diam E_{N_1} <\varepsilon$.
Thus, if $n \geq N_1$ and $m \geq N_1$ then $d(p_n,p_m) \leq \diam E_{N_1} < \varepsilon$. 
Hence $\{p_n\}$ is a Cauchy sequence.
\begin{theorem}\,
\begin{enumerate}
\item [(a)] If $\overline{E}$ is the closure of a set $E$ in a metric space $X$, then
\[
\diam \overline{E} = \diam E.
\]
\item[(b)] If $K_n$ is a sequence of compact sets in $X$ such that $K_n \supset K_{n+1}$ ($n=1,2,3,\ldots$) and if
\[
\lim_{n \to \infty} \diam K_n = 0,
\]
then $\cap_{1}^{\infty}K_n$ consists of exactly one point.
\end{enumerate}
\end{theorem}

\begin{theorem}\label{thm:compact_completeness} \,
\begin{enumerate}
\item[(a)] In any metric space $X$, every convergent sequence is a Cauchy sequence.
\item[(b)] If $X$ is a compact metric space and if $\{p_n\}$ is a Cauchy sequence in $X$, then $\{p_n\}$ converges to some point of $X$.
\item[(c)] In $\R^k$, every Cauchy sequence converges.
\end{enumerate}
\end{theorem}

\begin{definition}
A metric space in which every Cauchy sequence converges is said to be \emph{complete}.
\end{definition}

Thus Theorem \ref{thm:compact_completeness} says that \emph{all compact metric spaces and all Euclidean spaces are complete}.
Theorem \ref{thm:compact_completeness} implies also that \emph{every closed subset $E$ of a complete metric space $X$ is complete}. \footnote{This result is also immediate from definition.}

Theorem \ref{thm:limit_property}(c) shows that convergent sequences are bounded.
But bounded sequences need not converge.
However, there is one important case in which covnergence \emph{is} equivalent to boundness.

\begin{definition}
A sequence $\{s_n\}$ of real numbers is said to be

\begin{enumerate}
    \item[(a)] \emph{monotonically increasing} if $s_n \leq s_{n+1}$ ($n=1, 2, 3, \ldots$);
    \item[(b)] \emph{monotonically decreasing} if $s_n \geq s_{n+1}$ ($n=1, 2, 3, \ldots$). 
\end{enumerate}
\end{definition}

\begin{theorem}
Suppose $\{s_n\}$ is monotonic.
Then $\{s_n\}$ converges if and only if it is bounded.
\end{theorem}
